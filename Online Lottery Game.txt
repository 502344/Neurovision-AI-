Online Lottery Gameimport java.util.*;class Main {    public static void main(String[] args) {        Scanner sc=new Scanner(System.in);        int n=sc.nextInt();        int[]a=new int[n];        for(int i=0;i<n;i++){            a[i]=sc.nextInt();        }        Arrays.sort(a);            int max1=a[n-1];            int max2=a[n-2];            int min1=a[0];            int min2=a[1];            int prod1=max1*max2;            int prod2=min1*min2;            if(prod1>prod2){                System.out.println(max1+max2);            }            else{                System.out.println(min1+min2);            }    }}Output52 3 5 6 713
